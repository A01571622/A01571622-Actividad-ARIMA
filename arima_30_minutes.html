# Function for unit root tests with interpretation
def unit_root_tests(series, name):
    print(f"\nUnit Root Tests for {name}:")

    # ADF Test
    adf_result = adfuller(series)
    adf_stat = adf_result[0]
    adf_pvalue = adf_result[1]
    adf_critical = adf_result[4]
    adf_interpret = "Stationary" if adf_pvalue < 0.05 else "Non-stationary"
    adf_results.append({
        'Stock': name,
        'ADF Statistic': f"{adf_stat:.4f}",
        'ADF p-value': f"{adf_pvalue:.4f}",
        'ADF 5% Critical': f"{adf_critical['5%']:.4f}",
        'ADF Interpretation': adf_interpret
    })
    print("ADF Test:")
    print(f'ADF Statistic: {adf_stat:.4f}')
    print(f'p-value: {adf_pvalue:.4f}')
    print(f'Critical Values: {adf_critical}')
    print("Interpretation:")
    print(f"  - p-value {'<' if adf_pvalue < 0.05 else '>='} 0.05: "
          f"{'Reject' if adf_pvalue < 0.05 else 'Fail to reject'} null hypothesis - "
          f"{name} is {adf_interpret.lower()}")

    # KPSS Test
    kpss_result = kpss(series)
    kpss_stat = kpss_result[0]
    kpss_pvalue = kpss_result[1]
    kpss_critical = kpss_result[3]
    kpss_interpret = "Non-stationary" if kpss_pvalue < 0.05 else "Stationary"
    kpss_results.append({
        'Stock': name,
        'KPSS Statistic': f"{kpss_stat:.4f}",
        'KPSS p-value': f"{kpss_pvalue:.4f}",
        'KPSS 5% Critical': f"{kpss_critical['5%']:.4f}",
        'KPSS Interpretation': kpss_interpret
    })
    print("\nKPSS Test:")
    print(f'KPSS Statistic: {kpss_stat:.4f}')
    print(f'p-value: {kpss_pvalue:.4f}')
    print(f'Critical Values: {kpss_critical}')
    print("Interpretation:")
    print(f"  - p-value {'<' if kpss_pvalue < 0.05 else '>='} 0.05: "
          f"{'Reject' if kpss_pvalue < 0.05 else 'Fail to reject'} null hypothesis - "
          f"{name} is {kpss_interpret.lower()}")

# Function to generate random walk
def generate_random_walk(series, steps=30):
    # Calculate standard deviation of historical differences
    diffs = series.diff().dropna()
    std_dev = diffs.std()

    # Start random walk from last observed price
    last_price = series.iloc[-1]
    random_steps = np.random.normal(loc=0, scale=std_dev, size=steps)
    random_walk = [last_price]
    for step in random_steps:
        random_walk.append(random_walk[-1] + step)
    return pd.Series(random_walk[1:], index=range(len(series), len(series) + steps))

# Generate random walk series
amd_random_walk = generate_random_walk(amd_close, steps=30)
meta_random_walk = generate_random_walk(meta_close, steps=30)
adbe_random_walk = generate_random_walk(adbe_close, steps=30)

# Function to find best ARMA model (using ARIMA with d=0)
def find_best_arma(series, name, max_p=3, max_q=3):
    best_aic = float('inf')
    best_order = None

    for p in range(max_p + 1):
        for q in range(max_q + 1):
            try:
                model = ARIMA(series, order=(p, 0, q))
                results = model.fit()
                if results.aic < best_aic:
                    best_aic = results.aic
                    best_order = (p, 0, q)
            except:
                continue

    print(f"\nBest ARMA model for {name}:")
    print(f"Order: {best_order}")
    print(f"AIC: {best_aic:.2f}")
    print("Interpretation:")
    print(f"  - p={best_order[0]}: {best_order[0]} autoregressive term(s)")
    print(f"  - q={best_order[2]}: {best_order[2]} moving average term(s)")

    best_model = ARIMA(series, order=best_order).fit()
    return best_model

# Fit ARMA models for differenced series

meta_arma = find_best_arma(meta_diff, "META")

# Function to find best ARIMA model with interpretation
def find_best_arima(series, name, max_p=3, max_d=2, max_q=3):
    best_aic = float('inf')
    best_order = None

    for p in range(max_p + 1):
        for d in range(max_d + 1):
            for q in range(max_q + 1):
                try:
                    model = ARIMA(series, order=(p, d, q))
                    results = model.fit()
                    if results.aic < best_aic:
                        best_aic = results.aic
                        best_order = (p, d, q)
                except:
                    continue

    print(f"\nBest ARIMA model for {name}:")
    print(f"Order: {best_order}")
    print(f"AIC: {best_aic:.2f}")
    print("Interpretation:")
    print(f"  - p={best_order[0]}: {best_order[0]} autoregressive term(s)")
    print(f"  - d={best_order[1]}: {best_order[1]} difference(s) needed for stationarity")
    print(f"  - q={best_order[2]}: {best_order[2]} moving average term(s)")
    return best_order

# Find and fit best ARIMA models for original series

adbe_order = find_best_arima(adbe_close, "ADBE")

# Cointegration test with interpretation
def cointegration_test(df):
    result = coint_johansen(df, det_order=0, k_ar_diff=1)
    print("\nJohansen Cointegration Test:")
    print(f"Trace statistic: {result.lr1}")
    print(f"Critical values (90%, 95%, 99%): {result.cvt}")
    print("Interpretation:")
    coint_results = []
    for i in range(len(result.lr1)):
        trace_stat = result.lr1[i]
        crit_90, crit_95, crit_99 = result.cvt[i]
        interpret = ("Cointegration exists" if trace_stat > crit_95
                    else "No cointegration") + " at 95% confidence"
        coint_results.append({
            'Rank': f"r={i}",
            'Trace Statistic': f"{trace_stat:.2f}",
            '90% Critical': f"{crit_90:.2f}",
            '95% Critical': f"{crit_95:.2f}",
            '99% Critical': f"{crit_99:.2f}",
            'Interpretation': interpret
        })
        print(f"  - r = {i}: {interpret}")
        print(f"    Trace statistic ({trace_stat:.2f}) "
              f"{'>' if trace_stat > crit_95 else '<='} 95% critical value ({crit_95:.2f})")
    if result.lr1[0] > result.cvt[0, 1]:
        print("Conclusion: AMD, META, and ADBE are cointegrated - "
              "they share a long-run equilibrium relationship")
    else:
        print("Conclusion: No evidence of cointegration between AMD, META, and ADBE")
    return coint_results

}).dropna()

# Detailed forecast plot with historical data, forecast, confidence intervals, and random walk
def plot_forecast(model, series, random_walk, name, steps=30):
    forecast_obj = model.get_forecast(steps=steps)
    forecast = forecast_obj.predicted_mean
    conf_int = forecast_obj.conf_int()

    # Create continuous index for historical + forecast periods
    full_index = range(len(series) + steps)

    # Combine historical and forecast series
    historical_series = series.copy()
    forecast_series = pd.Series(forecast, index=range(len(series), len(series) + steps))

    # Plot historical and forecast in one continuous line
    plt.figure(figsize=(12,6))
    plt.plot(historical_series.index, historical_series, label=f'{name} Historical', color='blue')
    plt.plot(forecast_series.index, forecast_series, label=f'{name} ARIMA Forecast', color='red')

    # Plot confidence intervals for forecast period only
    plt.fill_between(forecast_series.index,
                     conf_int.iloc[:, 0],
                     conf_int.iloc[:, 1],
                     color='pink',
                     alpha=0.3,
                     label='95% Confidence Interval')

    # Plot random walk
    plt.plot(random_walk.index, random_walk, label=f'{name} Random Walk', color='green', linestyle='--')

    plt.title(f'{name} Price: Historical, ARIMA Forecast, and Random Walk')
    plt.xlabel('Time (30-minute intervals)')
    plt.ylabel('Price')
    plt.legend()
    plt.show()

    # Forecast interpretation
    last_value = series.iloc[-1]
    mean_forecast = forecast.mean()
    print(f"\nForecast Interpretation for {name}:")
    print(f"Last observed value: {last_value:.2f}")
    print(f"Average forecast value: {mean_forecast:.2f}")
    print(f"Forecast change: {mean_forecast - last_value:.2f}")
    if mean_forecast > last_value:
        print("Trend: Upward forecast trend")
    elif mean_forecast < last_value:
        print("Trend: Downward forecast trend")
    else:
        print("Trend: Flat forecast trend")
    print(f"95% CI range at period {steps}: [{conf_int.iloc[-1, 0]:.2f}, {conf_int.iloc[-1, 1]:.2f}]")

# 2. Correlograma (ACF and PACF Plots)
def plot_correlograma(series, name, lags=40):
    plt.figure(figsize=(12, 4))

    # ACF Plot
    plt.subplot(1, 2, 1)
    plot_acf(series, lags=lags, ax=plt.gca())
    plt.title(f'{name} ACF')

    # PACF Plot
    plt.subplot(1, 2, 2)
    plot_pacf(series, lags=lags, ax=plt.gca())
    plt.title(f'{name} PACF')

    plt.tight_layout()
    plt.show()
